// Bulón S.R.L. — Sistema de gestión de stock
// Esquema actual según la migración Flyway V1__esquema_inicial.sql

Project bulon_stock {
  database_type: 'SQL Server'
  Note: '''
    # Bulón S.R.L. — Sistema de gestión de stock
    Esquema versionado con Flyway (V1__esquema_inicial.sql).
    Hibernate solo valida el esquema (ddl-auto: validate).

    **Enums:** `tipo` y `rol` se guardan como VARCHAR con CHECK constraint;
    se modelan como Enum para que el diagrama muestre los valores posibles.
  '''
}

// ─────────────────────────── Enums ───────────────────────────

Enum tipo_movimiento {
  ENTRADA
  SALIDA
  AJUSTE
}

Enum rol {
  ADMINISTRADOR
  ENCARGADO_DEPOSITO
  VENDEDOR
}

// ─────────────────────────── Tablas ──────────────────────────

Table proveedores {
  id           bigint       [pk, increment]
  razon_social varchar(150) [not null]
  cuit         varchar(13)  [note: 'Único cuando no es NULL (índice filtrado uq_proveedores_cuit)']
  telefono     varchar(255)
  email        varchar(255)

  indexes {
    cuit [unique, name: 'uq_proveedores_cuit', note: 'WHERE cuit IS NOT NULL']
  }

  Note: 'Un proveedor tiene muchos productos.'
}

Table productos {
  id           bigint        [pk, increment]
  codigo       varchar(50)   [not null, unique, note: 'uq_productos_codigo']
  descripcion  varchar(200)  [not null]
  precio       numeric(12,2)
  stock_actual int           [not null, default: 0, note: 'CHECK >= 0. Solo cambia mediante movimientos']
  stock_minimo int           [not null, default: 0, note: 'CHECK >= 0. Umbral de alerta de reposición']
  proveedor_id bigint

  indexes {
    proveedor_id [name: 'ix_productos_proveedor']
  }

  Note: 'Requiere reposición cuando stock_actual <= stock_minimo.'
}

Table movimientos_stock {
  id          bigint          [pk, increment]
  producto_id bigint          [not null]
  tipo        tipo_movimiento [not null, note: 'varchar(20) + CHECK en la base']
  cantidad    int             [not null, note: 'Positiva en ENTRADA/SALIDA; puede ser negativa solo en AJUSTE']
  fecha       datetime2(6)    [not null, default: `SYSDATETIME()`]
  observacion varchar(255)

  indexes {
    (producto_id, fecha) [name: 'ix_movimientos_producto_fecha', note: 'fecha DESC — historial por producto']
  }

  Note: 'Historial de cambios de stock: da la trazabilidad del inventario.'
}

Table usuarios {
  id       bigint       [pk, increment]
  username varchar(50)  [not null, unique, note: 'uq_usuarios_username']
  nombre   varchar(100) [not null]
  rol      rol          [not null, note: 'varchar(30) + CHECK en la base']
  activo   bit          [not null, default: 1]

  Note: '''
    Todavía no se relaciona con ninguna tabla.
    Los movimientos no guardan qué usuario los registró.
  '''
}

// ───────────────────────── Relaciones ────────────────────────

Ref fk_productos_proveedor: productos.proveedor_id > proveedores.id
Ref fk_movimientos_producto: movimientos_stock.producto_id > productos.id
